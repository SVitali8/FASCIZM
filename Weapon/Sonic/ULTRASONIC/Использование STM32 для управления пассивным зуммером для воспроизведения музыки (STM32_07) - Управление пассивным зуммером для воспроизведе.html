I. Пассивные и активные зуммеры

Активный зуммер содержит источник колебаний и издает звук сразу после включения питания, но частота фиксирована, а тембр монотонный; пассивный зуммер не содержит источника колебаний, и его внутренняя структура эквивалентна электромагнитному громкоговорителю, который может издавать звук только путем подачи сигнала определенной частоты.

Человеческое ухо способно слышать частоты от 20 Гц до 20 кГц. Быстро переключаясь между высоким и низким уровнями выходного сигнала с помощью выводов GPIO микроконтроллера STM32, пассивный зуммер может издавать звук. Разные частоты соответствуют разной высоте тона.

#include "SysTick.h"

#include "misc.h"

u8 fac_us = 0;

u16 fac_ms = 0;

void SysTick_Init(u8 SYSCLK)

{

         SysTick_CLKSourceConfig(SysTick_CLKSource_HCLK_Div8);

         fac_us = SYSCLK / 8;

         fac_ms = (u16)fac_us*1000;

}

 

void delay_us(u32 nus)

{

         u32 temp;

         SysTick->LOAD = nus * fac_us;

         SysTick->VAL = 0;

         SysTick->CTRL |= SysTick_CTRL_ENABLE_Msk;

         делать{

                   temp = SysTick->CTRL;

         }while((temp&0x01)&&(!(temp&(1<<16))));

         SysTick->CTRL &= ~SysTick_CTRL_ENABLE_Msk;

         SysTick->VAL = 0;

}

void delay_ms(u16 nms)

{

         u32 temp;

         SysTick->LOAD = nms * fac_ms;

         SysTick->VAL = 0;

         SysTick->CTRL |= SysTick_CTRL_ENABLE_Msk;

         делать{

                   temp = SysTick->CTRL;

         }while((temp&0x01)&&(!(temp&(1<<16))));

         SysTick->CTRL &= ~SysTick_CTRL_ENABLE_Msk;

         SysTick->VAL = 0;

}

Эти два файла реализуют функции для точной задержки с помощью SysTick, которые затем передаются последующим программам генерации звуковых частот.
